\begin{abstract}
We present a concise statement of the Absolute Foundation No-Go Theorem (AFN-T) and its boundary mechanisms in a specified metatheory $S \in \RS$: (i) for direct or slice-conservative presentations in the selected derived class $\cS_S$, the identity functor is an interpretation into $\cS_S$, refuting the non-reducibility predicate $\Pi_4$; (ii) for arbitrary faithful subtheory inclusions $\iota_F : F \hookrightarrow S$ equipped with a canonical $\Sigma_1$-binumeration and classical (or negative-translated) relative proof-translation, testing maximal generativity $\Pi_{3\text{-max}}$ against $T_S = \PA + \Con_{\sigma_S}(S)$ refutes $(F_S) \wedge (\Pi_{3\text{-max}})$ via Feferman--Pudl\'ak Second Incompleteness. Neither mechanism excludes new foundations or operational refinements. We distinguish these boundary results from conditional classifier-comparison criteria, fragment-level intensional separation (\texttt{thm:I-fragment}), and sharp epistemic splits across the six frontier obligations \textup{(B1)--(B6)}: finite-rank unstratified meta-closure and $(\mathrm{Stab}\text{-}2')$ for $\Cat_{(\infty,\omega)}^{\tau\text{-}\mathrm{lim}}$ are refuted, $\Cat(\Cat_{(\infty,\omega)}^{\mathrm{coind, GH}}) \simeq \Cat_{(\infty,\omega)}^{\mathrm{coind, GH}}$ and strict $2$-categorical Pronk/\texttt{Perf} hold constructively, while full $(\infty,2)$-hyperstack existence (\texttt{131.T}), global \textup{(I-4)}, unbounded meta-stabilization, and exact consistency-strength determinations remain open construction obligations. This summary matches the main MSFS paper.

\medskip
\noindent\textbf{Keywords:} no-go theorems, categorical logic, $(\infty, n)$-categories, Lambek--Scott semantics, meta-classification.

\noindent\textbf{MSC2020:} 03A05, 03B30, 18A05, 18N10, 18N50, 18N60, 18N65.
\end{abstract}

\section{Introduction}\label{sec:intro}

Classical results of Cantor, Russell, G\"odel, Tarski, and Lawvere have distinct hypotheses, domains, and recursion- or set-theoretic inputs. They serve as motivating analogues rather than strict corollaries of the boundary lemma below (indeed, the diagonal branch $(\alpha_{\mathrm{diag}})$ explicitly invokes the Feferman--Pudl\'ak form of G\"odel's Second Incompleteness Theorem as an input).

\paragraph{Main result.} For a direct or slice-conservative presentation in $\cS_S$, the identity reduces the candidate to itself, refuting $\Pi_4$; for an arbitrary faithful subtheory inclusion $\iota_F : F \hookrightarrow S$ under canonical $\Sigma_1$-binumeration and classical (or negative-translated) relative interpretability, diagonalization via $T_S = \PA + \Con_{\sigma_S}(S)$ refutes $(F_S) \wedge (\Pi_{3\text{-max}})$ (Theorem~\ref{thm:afnt}). No full faithfulness of arbitrary syntax-to-model evaluation is claimed.

\paragraph{Scope.} Both arguments apply at each constructed categorical or metatheoretic stage where their hypotheses hold. They do not construct coherent higher-depth models or an unbounded tower of universes.

\paragraph{Comparison and refinement.} Theorems~\ref{thm:meta-cat} and~\ref{thm:meta-mult} give conditional comparison criteria. Theorem~\ref{thm:intensional} supplies an ordinary slice construction, whose equivalence-only display filtration is trivial. Further intensional data require their own construction and preservation proofs.

\paragraph{Open obligations.} Remarks~\ref{thm:stab}, \ref{thm:reflective}, and~\ref{thm:meta-stab} record open stabilization and strength obligations alongside proved fragment/obstruction splits across \textup{(B1)--(B6)}. No exact equiconsistency with two inaccessible cardinals follows from choosing that ambient base. In the companion verification corpora, the standalone Verum Wave~1+2 modules (\texttt{13} zero-axiom modules, \texttt{55} multi-step theorems, \texttt{592/592} proved) plus Track~B correspondence proofs (\texttt{proofs/ss-membership/}, \texttt{proofs/afnt-alpha-id/}) complement the constructive Lean~4 kernel \texttt{lean4-mf-kernel} (\texttt{82} modules, \texttt{25{,}777} declarations, \texttt{915} audited \texttt{\#print axioms} checks, \texttt{267} zero-axiom theorems, \texttt{115} \texttt{NonVacuity} instances, \texttt{0} \texttt{Classical.choice}), while \texttt{43} legacy schematic axioms remain quarantined.

\paragraph{Conventions.} We work throughout in $\ZFC$ augmented by two inaccessible cardinals $\kappa_1 < \kappa_2$, providing Grothendieck universes $\mathbf{U}_1 \subset \mathbf{U}_2$. Sufficiency of this base for each proposed construction requires its own size and existence checks. The categorical machinery follows Kelly~\cite{Kelly1982} for enriched and $2$-categories, Lurie~\cite{LurieHTT, LurieHA} for $(\infty, 1)$-categories, Riehl--Verity~\cite{RiehlVerity2022} for $\infty$-cosmoi, Ad\'amek--Rosick\'y~\cite{AdamekRosicky1994} for accessible functors, and Barwick--Schommer-Pries~\cite{BarwickSchommerPries2021} for finite-$n$ unicity.

\section{\texorpdfstring{Rich-Metatheories and the Class $\cS_S$}{Rich-Metatheories and the Class S-S}}\label{sec:rs}

\begin{definition}[Rich-metatheory]\label{def:rs}
A theory $S$ formulated in a first-order meta-syntax is a \textbf{Rich-metatheory} ($\RS$) if:
\begin{itemize}
\item \textbf{(R1)} $S$ interprets Robinson arithmetic $\mathsf{Q}$.
\item \textbf{(R2)} The set of axioms of $S$ is recursively enumerable over a countable language.
\item \textbf{(R3)} $S$ has a model $\cM$ in the outer Grothendieck universe $\mathbf U_2$.
\item \textbf{(R4)} There is a recursive injection $\ulcorner \cdot \urcorner : \mathrm{Formulas}(S) \to \mathbb{N}$ such that the predicates ``$n$ codes a provable formula'' and ``$n$ codes a well-formed formula'' are $S$-definable.
\item \textbf{(R5)} There is $n_S \in \mathbb{N} \cup \{\infty\}$ such that the syntactic category $\Syn(S)$ (contexts, provable-from-context substitutions, higher homotopies) is an essentially $\mathbf{U}_1$-small $(\infty, n_S)$-category whose $\mathrm{Ind}_{\kappa_S}$-completion ($\kappa_S \in \{\kappa_1, \kappa_2\}$) is $\kappa_S$-accessible, and for every model $\cM \in \mathbf{U}_2$ of $S$ the evaluation $\ev : \Syn(S) \to \cM$ is an $S$-preserving $(\infty, n_S)$-functor participating in a Lambek--Scott-type syntax--semantics adjunction $\Syn \dashv \Mod$.
\end{itemize}
We denote the class of Rich-metatheories as $\RS$, and the $2$-category of Rich-foundations ($\LFnd$) as $\cF$.
\end{definition}

Examples must be checked against (R1)--(R5) using a specific formal theory and a chosen categorical semantics. A mathematical programme is not itself a recursive axiomatization, and an evaluation functor into a single model is neither automatically fully faithful nor automatically a right adjoint. The general higher-level syntax--semantics bridge is a hypothesis to verify in each application.

\begin{definition}[The selected presentation class $\cS_S$]\label{def:SS}
Fix a syntactic presentation $\Syn(S)$ and the ambient universe $\mathbf{U}_2$. Define $\cS_S^{\mathrm{local}} \subseteq \mathbf{U}_2$ to contain the pointed syntactic presentations $(\Syn(S), [\phi])$ for admissible formulas $\phi$ of $S$ (essentially $\mathbf{U}_1$-small) together with pointed model objects $(\cM, Y)$ for $\cM \in \mathbf{U}_2$, $\cM \models S$, $Y \in \Ob(\cM)$. Define $\cS_S^{\mathrm{global}} \subseteq \mathbf{U}_2$ by closing $\cS_S^{\mathrm{local}}$ under the fixed $\mathbf{U}_2$-bounded derived categorical operations (Kan extensions, Grothendieck constructions, $\mathbf{U}_2$-small limits/colimits when they exist in $\mathbf{U}_2$, slice/comma/opposite categories, derived functors, monoidal products/internal homs). Set $\cS_S := \cS_S^{\mathrm{local}} \cup \cS_S^{\mathrm{global}}$. Thus $\cS_S$ is a $\mathbf{U}_2$-class ($\subseteq \mathbf{U}_2$, of size $\le \kappa_2$); restricting both model carriers and diagram indices to $\mathbf{U}_1$ yields a $\mathbf{U}_2$-small fragment $\cS_S^{(\mathbf{U}_1)} \in \mathbf{U}_2$. Accessibility of derived objects is verified case by case rather than assumed globally.
\end{definition}

\begin{lemma}[Membership of the chosen syntactic presentation]\label{lem:SS-membership}
If a formally described candidate is represented directly by the pointed syntactic presentation $(\Syn(S), [\phi_X])$ for $\phi_X \in S$, then that presentation belongs to $\cS_S^{\mathrm{local}} \subseteq \cS_S$.
\end{lemma}
\begin{proof}
It is included in $\cS_S^{\mathrm{local}}$ by Definition~\ref{def:SS}.
\end{proof}
When $\phi_X$ is instead presented inside a proper subtheory $\iota_F : F \hookrightarrow S$, identifying $(\Syn(F), [\phi_X]_F)$ with $(\Syn(S), [\iota_F(\phi_X)]_S)$ requires slice-conservativity of $\Syn(\iota_F)$ (which fails for non-conservative extensions such as $\PA \hookrightarrow \PA + \Con(\PA)$). Theorem~\ref{thm:afnt} handles both the slice-conservative case and arbitrary faithful subtheories $\iota_F : F \hookrightarrow S$.

\begin{definition}[Morita-reducibility]\label{def:morita}
An $(\infty, n)$-functor $F : X \to Y$ is an \emph{$(\infty, n)$-equivalence onto its image} if $F$ is fully faithful and essentially surjective onto the full sub-$(\infty, n)$-category of $Y$ given by $\image(F)$. We say $X$ is \emph{$(\infty, n)$-Morita-reducible} to $Y$, written $X \to_{M, n} Y$, if such a functor exists.
\end{definition}

\section{The Absolute Foundation No-Go Theorem}\label{sec:afnt}

\begin{theorem}[AFN-T $(\alpha)$-part]\label{thm:afnt}
Fix $S \in \RS$ and $n \in \mathbb{N} \cup \{\infty\}$, and let $X$ be a pointed $(\infty, n)$-categorical candidate presented via a faithful subtheory inclusion $\iota_F : F \hookrightarrow S$ and $(\Syn(F), [\phi_X]_F) \simeq_{(\infty, n)} X$ (condition $(F_S)$):
\begin{enumerate}[nosep,label=(\roman*)]
\item \emph{Direct / slice-conservative mechanism $(\alpha_0)$}: whenever $F = S$ or $\Syn(\iota_F)|_{[\phi_X]_F} : (\Syn(F), [\phi_X]_F) \to (\Syn(S), [\iota_F(\phi_X)]_S)$ is an $(\infty, n)$-equivalence onto its image, $X$ reduces to an element of $\cS_S^{\mathrm{local}} \subseteq \cS_S$ and therefore cannot satisfy
\[
 (\Pi_{4, S, n}):\quad X \not\to_{M, n} Y \quad\text{for every } Y \in \cS_S.
\]
\item \emph{Strong diagonal anti-absoluteness $(\alpha_{\mathrm{diag}})$}: suppose $S$ is equipped with a canonical $\Sigma_1$-binumeration $\sigma_S$ of its r.e.\ axiom set (fixing $\Con_{\sigma_S}(S)$), and every composite interpretation $J := \iota_F \circ \tau : \PA + \Con_{\sigma_S}(S) \to S$ arising from maximal generativity $(\Pi_{3\text{-max}, S, n})$ is a classical first-order relative interpretation (or is accompanied by an explicit negative translation and primitive-recursive proof-translation verifiable in $\PA$). Then for every faithful inclusion $\iota_F : F \hookrightarrow S$, the generative pair $(F_S) \wedge (\Pi_{3\text{-max}, S, n})$ is inconsistent.
\end{enumerate}
In particular, across all $(S, n)$ in the scope of (i) or (ii), the $\LAbs$ conjunction $(F_S) \wedge (\Pi_{4, S, n}) \wedge (\Pi_{3\text{-max}, S, n})$ is empty.
\end{theorem}
\begin{proof}
For (i), take $Y := (\Syn(S), [\iota_F(\phi_X)]_S) \in \cS_S^{\mathrm{local}} \subseteq \cS_S$ (for $F = S$, $Y = X$ via $\id_X$); the induced equivalence onto its image violates $\Pi_4$. For (ii), by (R3) the theory $S$ is syntactically consistent in standard $\mathbb{N}$, so $T_S := \PA + \Con_{\sigma_S}(S)$ has standard model $\mathbb{N} \in \mathbf{U}_1 \subset \mathbf{U}_2$ and lies in $\cF$. Applying $(\Pi_{3\text{-max}, S, n})$ at $T_S$ and composing with $\iota_F$ yields $J : \PA + \Con_{\sigma_S}(S) \to S$. Its classical arithmetic pullback $S_{\mathrm{arith}}$ is a consistent r.e.\ extension of $\PA + \Con_{\sigma_S}(S)$ in which primitive-recursive proof-translation gives $S_{\mathrm{arith}} \vdash \Con_{\sigma_{\mathrm{arith}}}(S_{\mathrm{arith}})$, contradicting G\"odel--Feferman--Pudl\'ak Second Incompleteness (Feferman~\cite{Feferman1960}, Thm.~5.6; Pudl\'ak~\cite{Pudlak1985}, Thm.~2.1).
\end{proof}
Unlike an inference of full faithfulness from syntactic initiality, neither mechanism assumes that an arbitrary evaluation functor into a model is fully faithful.

\begin{theorem}[Limit exclusion; AFN-T $(\beta)$-part]\label{thm:afnt-beta}
Let $A_\infty := \colim_{\kappa < \lambda} A_\kappa$ be the colimit of a transfinite diagram in the ambient $(\infty, n)$-categorical meta-universe:
\begin{enumerate}[nosep,label=(\alph*)]
\item \emph{Definable $\mathbf{U}_2$-cofinal closure}: if $\mathrm{cf}(\lambda) < \kappa_2$, a cofinal $\mathbf{U}_2$-small subdiagram $D : I \to \cS_S^{(k)}$ has $S$-definable objects and coherent transition functors, and $A_\infty$ exists in $\mathbf{U}_2$, then $A_\infty \in \cS_S^{(k+1)} \subseteq \cS_S^{\mathrm{global}}$ and fails $(\Pi_{4, S, n})$.
\item \emph{Unconditional $\LAbs$ dichotomy}: for any directed or proper-class diagram without assuming $S$-definability of transitions or fixed-cardinal accessibility closure, excluded middle splits on $\exists S' \in \RS\,(F_{S'})(A_\infty)$: either $A_\infty$ satisfies $(F_{S'})$ for some $S' \in \RS$ (refuted by Theorem~\ref{thm:afnt}), or it fails $(F_{S'})$ for all $S' \in \RS$. Thus $A_\infty \notin \LAbs$ within the scope of Theorem~\ref{thm:afnt}.
\end{enumerate}
\end{theorem}
\begin{proof}
Part (a) applies the $\mathbf{U}_2$-small colimit closure of Definition~\ref{def:SS} and Theorem~\ref{thm:afnt}~(i). Part (b) applies excluded middle in $\ZFC + 2\text{-inacc}$ directly to the limit object $A_\infty$, avoiding any invalid reconstruction of diagram uniformity from a specification of $A_\infty$.
\end{proof}
A bound on cofinality alone neither constructs an $S$-definable diagram nor guarantees preservation of a fixed accessibility cardinal $\kappa_1$.

\section{Absoluteness}\label{sec:absolute}

\begin{theorem}[Transfer of the boundary obstruction]\label{thm:absolute}
At every stage, depth, or faithfully embedded categorical doctrine where a candidate satisfies the hypotheses of Theorem~\ref{thm:afnt}~(i) or~(ii), the conclusion of Theorem~\ref{thm:afnt} holds.
\end{theorem}
\begin{proof}
Apply Theorem~\ref{thm:afnt} in that presentation.
\end{proof}
Transferring a claim between different metatheories, depths, or semantic doctrines (such as double or fusion categories) requires an explicitly supplied translation or faithful embedding preserving the boundary predicates. This does not establish exhaustiveness of all possible variations.

\begin{remark}[Unbounded stabilization: coinductive fixed point vs.\ truncation-limit refutation]\label{thm:stab}
Barwick--Schommer-Pries~\cite{BarwickSchommerPries2021} compares models of $(\infty,n)$-categories for fixed finite $n$ (with automorphism space $B(\mathbb{Z}/2)^n$). In the main MSFS paper, $\Cat_{(\infty,\omega)}^{\mathrm{bdd}}$ lacks countable coproducts and disagrees with both $\Cat_{(\infty,\omega)}^{\mathrm{coind}}$ and $\Cat_{(\infty,\omega)}^{\tau\text{-}\mathrm{lim}}$; moreover, \textup{(Stab-2')} is unconditionally refuted for $\Cat_{(\infty,\omega)}^{\tau\text{-}\mathrm{lim}}$ (yielding $|\pi_0\mathrm{Map}| \ge 2^{\aleph_0} \neq 1$), whereas the Gepner--Haugseng inductive coinductive limit satisfies $\Cat(\Cat_{(\infty,\omega)}^{\mathrm{coind, GH}}) \simeq \Cat_{(\infty,\omega)}^{\mathrm{coind, GH}}$ unconditionally (and arbitrary BSP21 models conditionally on coherent $(\mathbb{Z}/2)^n$-orientations). Three barriers remain open: BSP21 non-unicity at $n = \infty$, external hypercompleteness \textup{(Stab-1)}, and bridging $\Cat_{(\infty,\omega)}^{\mathrm{coind, GH}}$ to $\fM$ and $n_S = \infty$ syntax.
\end{remark}

\section{Closure of Bypass Paths}\label{sec:bypass}

The following statements specify what can be concluded for constructed universe families, reflection sequences, and intensional data. They do not exclude every possible extension of these kinds.

\subsection{Universe Polymorphism}

\begin{theorem}[Definable coherent universe families]\label{thm:universe}
Let $D:I\to\Cat$ be a specified small $S$-definable diagram. If its relevant limit exists within the chosen universe and is included among the allowed derived constructions, that limit lies in $\cS_S$ and fails $\Pi_4$.
\end{theorem}
\begin{proof}
Use the closure hypothesis and Theorem~\ref{thm:afnt}.
\end{proof}
An endofunctor of a larger universe need not preserve a smaller one. Thus restriction does not automatically define transition functors between universe-endofunctor categories. Their preservation and coherence must be proved. An individual compatible family is also not identical to the entire category of compatible families.

\subsection{Reflective Towers}

\begin{remark}[Reflection requires a separate strength analysis]\label{thm:reflective}
The rule $S_{\alpha+1}=S_\alpha+\Con(S_\alpha)$ does not specify limit stages, an effective ordinal-notation system, or a uniform reflection principle. No general statement that reflection adds exactly one inaccessible or stabilizes at an expression $\Con(S)+\kappa$ is established. At each actually constructed stage, the relative identity obstruction remains available with that stage's own derived class; it does not compare proof-theoretic strengths or show termination.
\end{remark}

\subsection{Intensional Refinement}

\begin{definition}[Display data and the ordinary target]\label{def:Sint}
For the verified ordinary construction, a display category is a category with pullbacks and a selected class of arrows containing isomorphisms and closed under pullback, composition, and isomorphism of arrows. Let $\Sint$ be their $2$-category with display- and pullback-preserving functors and natural transformations. A higher target requires a separately specified model of arrows, transformations, and coherence.
\end{definition}

\begin{theorem}[Ordinary slice construction]\label{thm:intensional}
For an ordinary locally small category $\mathcal B$ with chosen pullbacks, the assignment $F\mapsto(\mathcal B/F,\text{isomorphisms})$, with base-change functors, defines a contravariant pseudofunctor $\mathcal B^{\mathrm{op}}\to\Sint$. Its display filtration generated from isomorphisms by the stated operations is constant at stage zero.
\end{theorem}
\begin{proof}
Slices have pullbacks and base change preserves them. Their universal property gives coherent identity and composition isomorphisms for reindexing. Every generating operation preserves isomorphisms, so it cannot add a nonisomorphism to the seed class.
\end{proof}
On the concrete syntactic fragment $\mathcal{F}_{\MLTT}$ ($\MLTT_{\mathrm{UIP}+\mathrm{FE}}$ vs.\ $\ETT$~\cite{KapulkinLi2025}), the main paper constructively discharges \textup{(I-1)--(I-3)} via directed $\beta\eta$-reduction $2$-cells \textup{(W2)} (invariant under arbitrary $2$-equivalences in $\mathcal{S}_{\mathrm{int}}^{\mathrm{strict}}$), while non-collapsing gradings are realized both by operational silent-step budgets (\texttt{ActCostGrade}) and by intrinsic fiber-chain heights in $\mathbf{Poset2Cat}_{\mathbb{Z}_2}$ ($\operatorname{gr}_G(p_{2^{2^n}}) = n$). Global \textup{(I-4)} splits into proved bivariant bridges, sharp variance/pullback/implication-collapse/Pronk obstructions (\textup{(I-4a)--(I-4c)}), and remaining open obligations (non-definitional $\cF$-bivariance, full impredicative $\ZFC$ $2$-cells, and global $\Sint^{\RS}[\mathcal{W}_{\mathrm{Morita}}^{-1}] \simeq \fM$).

Given a comparison $\widetilde\pi I\simeq\pi$ with compatible domains, the base image of $I(F)$ is $\pi(F)$. A fibre must include its comparison-to-the-base data; nontriviality and a fibration structure are additional claims. Such base-image constancy does not prove absence of new computational or resource-sensitive structure.

\paragraph{Finite observation and search.}
An executable observational quotient fixes the interpreter, complete library
meanings, all admission evidence, and the continuation policy. Exact register
traces with a separate undefined outcome support a continuation congruence
only with all policy metadata, including constant-use and selection flags,
and equality-preserving updates. This preserves the specified finite checks,

\section{\texorpdfstring{Meta-Classification of $\LCls$ Frameworks}{Meta-Classification of L\_Cls Frameworks}}\label{sec:meta}

\begin{definition}[$\LCls$ meta-structure]\label{def:meta}
A $2$-category $\mathbf{F}$ is an \emph{$\LCls$ meta-structure} if:
\begin{itemize}
\item \textbf{(M1)} locally small;
\item \textbf{(M2)} equipped with a classification $2$-functor $\Cl_{\mathbf{F}} : \mathbf{F} \to \fM(\mathbf{F}) \subseteq \fM$;
\item \textbf{(M3)} accessible reflection $R_{\mathbf{F}} : \mathbf{F} \to \mathbf{F}$;
\item \textbf{(M4)} non-generative: $\mathbf{F}$ does not derive the axioms of $F \in \image(\Cl_{\mathbf{F}})$;
\item \textbf{(M5)} parametrized by some $S \in \RS$.
\end{itemize}
$\mathbf{F}$ is \emph{maximal} (and we write $\mathbf{F} \in \Meta_{\mathrm{Cls}}^{\top}$) if additionally:
\begin{itemize}
\item \textbf{(Max-1)} $\image(\Cl_{\mathbf{F}}) = \fM$;
\item \textbf{(Max-2)} $\mathrm{Aut}_2(\mathbf{F})$ acts transitively on gauge classes in $\image(\Cl_{\mathbf{F}})$;
\item \textbf{(Max-3)} $\mathbf{F}$ admits a depth-indexed filtration $\{\mathbf{F}^{(m)}\}_{m < \omega}$ with $\mathbf{F} \simeq_2 \colim_m \mathbf{F}^{(m)}$ and a depth function $\mathrm{depth} : \Ob(\mathbf{F}) \to \omega$ such that every comprehension-defined object of depth $m$ uses only objects of depth $< m$;
\item \textbf{(Max-4)} a correctly typed higher intensional construction satisfying the additional (I-1)--(I-4) obligations after Theorem~\ref{thm:intensional}, with the joint-separation condition of the main paper. Its existence is not supplied by the ordinary slice construction.
\end{itemize}
The class of $\LCls$ meta-structures is $\Meta_{\mathrm{Cls}}$; the class of maximal meta-structures is $\Meta_{\mathrm{Cls}}^{\top} \subseteq \Meta_{\mathrm{Cls}}$.
\end{definition}

\begin{theorem}[Conditional comparison criterion]\label{thm:meta-cat}
Suppose $2$-functors $\Phi_i:\mathbf F_i\to\mathbf K$, $i=1,2$, are essentially surjective on objects and induce equivalences on all hom-categories. Then $\mathbf F_1\simeq_2\mathbf F_2$.
\end{theorem}
\begin{proof}
Both comparisons are biequivalences. Compose one with a pseudoinverse of the other.
\end{proof}
The maximality axioms do not by themselves supply these comparisons or lift arbitrary morphisms. For example, $1\to BG$ for nontrivial $G$ is essentially surjective and faithful but not full. Models of an accessible theory need not all be equivalent; the theory of groups already has inequivalent models. A higher-depth equivalence requires further coherence data.

\begin{theorem}[Conditional image separation]\label{thm:meta-mult}
Two constructed classifiers over the same base with distinct essential classification images cannot be equivalent over that base. A classifier with a proper image fails (Max-1).
\end{theorem}
\begin{proof}
An equivalence commuting with the classification functors preserves their essential images.
\end{proof}
Applying this criterion to Univalent Foundations, $\infty$-cosmoi, or cohesive frameworks requires actual classifiers satisfying (M1)--(M5) and proofs of the proposed image distinctions. Programme names and different syntactic presentations do not furnish these witnesses.

\begin{remark}[Meta-classification: proved obstructions, conditional idempotency, and open core]\label{thm:meta-stab}
The main paper establishes finite-stage existence of $\fM^{(\mathrm{Cls})}_{\mathbf{U}_1} \in \mathbf{U}_2$ alongside automatic \textup{(M2)/(M3)} obstructions (ill-defined $[S_{\mathbf{F}}]$ and closure reflectors $R'_\mu$ on $\mathcal{P}(\mu^+)$), unconditional stage separation and independence of $\mathbf{U}_3$ and $\mathbf{U}_1 \prec_{\Sigma_2} \mathbf{U}_2$ from $\ZFC + 2\text{-inacc}$, conditional idempotency $\fM^{(\mathrm{Cls}^2)}_\lambda \simeq_2 \fM^{(\mathrm{Cls})}_\lambda$ under \textup{(Idem-1)--(Idem-4)} (with $L : \Meta_{\mathrm{Cls}}^{(\lambda)} \rightleftarrows \Meta_{\mathrm{Cls}}^{(\lambda,2)} : U$, $\Sigma_2$-reflection consistency bracket strictly below Mahlo, and mutual independence of all four \textup{Idem} clauses), and an unconditional refutation of unstratified finite-rank closure inside any fixed $\mathbf{U}_n$. Likewise, Grothendieck-fibered quasi-categoricity holds conditionally under \textup{(Fib-1)--(Fib-3)} (mutually independent), while the full $(\infty, 2)$-hyperstack problem \texttt{131.T} (\textup{(B3-Obs1)--(B3-Obs4)}) and unbounded/universal meta-stabilization remain open. At every stage actually constructed, Theorem~\ref{thm:afnt} still applies independently of stabilization.
\end{remark}

\section{Consistency Strength}\label{sec:cons}

\begin{theorem}[Conditional definitional conservativity]\label{thm:consistency}
If all new apparatus is introduced by explicit definitions whose existence and uniqueness are proved in a fixed base $B$, adjoining its names is conservative over $B$ for sentences of the old language.
\end{theorem}
\begin{proof}
Eliminate the defined names throughout any proof, using existence and uniqueness for functional definitions.
\end{proof}
This definitional conservativity \textup{(O-AC5a)} separates sharply from equiconsistency \textup{(O-AC5b)}: categorical localization equivalence $\fM \simeq \fM_{\mathcal{E}}$ does not entail $\Con(\mathcal{T}_{\mathrm{ext}}) = \Con(\mathcal{T}_{\mathrm{base}})$ (refuted via $(\ZFC, \ZFC+2\text{-}\mathrm{inacc})$ and $\PA + \Sigma_{\mathrm{refl}}$ by G\"odel~II). A complete model construction gives an upper bound; equiconsistency additionally requires a reverse interpretation or a separate lower-bound proof.

\begin{corollary}[Emptiness of $\LAbs$]\label{cor:level-6}
Within the scope of Theorem~\ref{thm:afnt} (every direct or slice-conservative presentation over $\RS$, and every faithful subtheory presentation satisfying the diagonalization hypotheses of Theorem~\ref{thm:afnt}~(ii)), no candidate satisfies $(F_S) \wedge (\Pi_{4, S, n}) \wedge (\Pi_{3\text{-max}, S, n})$, so $\LAbs = \emptyset$.
\end{corollary}

\begin{proof}
Immediate from Theorem~\ref{thm:afnt}.
\end{proof}
